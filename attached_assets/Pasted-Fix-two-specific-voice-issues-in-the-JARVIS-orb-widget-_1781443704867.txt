WAKE WORD LISTENER  (startWakeWordListener function)
──────────────────────────────────────────────────────

const SR = window.SpeechRecognition || window.webkitSpeechRecognition;

function startWakeWordListener() {
  const recognition = new SR();
  recognition.continuous = true;
  recognition.interimResults = true;
  recognition.lang = voiceSettings.lang; // default 'en-US'

  recognition.onresult = (event) => {
    const transcript = Array.from(event.results)
      .map(r => r[0].transcript)
      .join(' ')
      .toLowerCase()
      .trim();

    // Wake phrases — any of these trigger activation
    const wakePhrases = [
      'hello jarvis',
      'hey jarvis',
      'ok jarvis',
      'jarvis',         // fallback single word
    ];

    const triggered = wakePhrases.some(phrase =>
      transcript.includes(phrase)
    );

    if (triggered) {
      recognition.stop();           // stop wake listener
      setOrbStatus('listening');    // orb turns gold
      openOrbChat();                // open chat panel if closed

      // Extract query after the wake phrase if present
      // e.g. "Hello Jarvis what is the weather" → "what is the weather"
      let query = transcript;
      wakePhrases.forEach(phrase => {
        query = query.replace(phrase, '').trim();
      });

      if (query.length > 2) {
        // User said wake word + query in one breath
        // wait 400ms then send directly
        setTimeout(() => {
          sendVoiceMessage(query);
        }, 400);
      } else {
        // Wake word only — start fresh listening for the query
        startQueryListening();
      }
    }
  };

  recognition.onend = () => {
    // Auto-restart wake listener unless deliberately stopped
    if (wakeWordActive && !isListeningForQuery) {
      setTimeout(() => startWakeWordListener(), 300);
    }
  };

  recognition.onerror = (e) => {
    if (e.error === 'no-speech') {
      // Normal — just restart
      setTimeout(() => startWakeWordListener(), 300);
    } else if (e.error === 'not-allowed') {
      setWakeWordBlocked(true);
    }
  };

  recognition.start();
  wakeRecognitionRef.current = recognition;
}

QUERY LISTENING  (startQueryListening function)
──────────────────────────────────────────────────────

Called after wake word is detected with no trailing query.
Listens for a single utterance then sends it.

function startQueryListening() {
  const recognition = new SR();
  recognition.continuous = false;
  recognition.interimResults = true;
  recognition.lang = voiceSettings.lang;

  // Show interim transcript in chat input as user speaks
  recognition.onresult = (event) => {
    const interim = Array.from(event.results)
      .map(r => r[0].transcript)
      .join('');
    setInputPreview(interim); // show grey preview text in input
  };

  recognition.onend = () => {
    const final = finalTranscriptRef.current.trim();
    if (final.length > 1) {
      // Auto-send after delay (user can cancel)
      showCancelCountdown(voiceSettings.autoSendDelay, () => {
        sendVoiceMessage(final);
      });
    } else {
      // Nothing heard — return to wake listener
      setOrbStatus('online');
      startWakeWordListener();
    }
    isListeningForQuery = false;
  };

  recognition.onerror = () => {
    setOrbStatus('online');
    startWakeWordListener();
    isListeningForQuery = false;
  };

  isListeningForQuery = true;
  recognition.start();
  queryRecognitionRef.current = recognition;
}

FIX 2 — TTS FIRES ON EVERY AI RESPONSE
═══════════════════════════════════════════════════════

In VoiceController.tsx, after every AI message is received
(both from orb chat and from wake word flow), automatically
speak the response using SpeechSynthesis.

──────────────────────────────────────────────────────
speakResponse function
──────────────────────────────────────────────────────

function speakResponse(text: string) {
  if (!('speechSynthesis' in window)) return;

  // Cancel any in-progress speech
  window.speechSynthesis.cancel();

  // Clean the text
  const clean = text
    .replace(/#{1,6}\s/g, '')        // headings
    .replace(/\*\*(.*?)\*\*/g, '$1') // bold
    .replace(/\*(.*?)\*/g, '$1')     // italic
    .replace(/`(.*?)`/g, '$1')       // inline code
    .replace(/```[\s\S]*?```/g, 'code block omitted')
    .replace(/\[(.*?)\]\(.*?\)/g, '$1') // links
    .replace(/\$/g, 'dollars')
    .replace(/%/g, 'percent')
    .replace(/\bms\b/g, 'milliseconds')
    .replace(/\btok\b/g, 'tokens')
    .replace(/\n+/g, '. ')
    .trim()
    .slice(0, voiceSettings.maxSpeakLength);

  const utterance = new SpeechSynthesisUtterance(clean);

  // Voice selection
  const voices = window.speechSynthesis.getVoices();
  const preferred = [
    'Google UK English Male',
    'Microsoft Ryan Online (Natural) - English (United Kingdom)',
    'Microsoft Guy Online (Natural) - English (United States)',
    'Daniel',
    'Alex',
  ];
  let selectedVoice = null;
  for (const name of preferred) {
    selectedVoice = voices.find(v => v.name === name) || null;
    if (selectedVoice) break;
  }
  // Fallback: first English male-sounding voice
  if (!selectedVoice) {
    selectedVoice = voices.find(v => v.lang.startsWith('en')) || null;
  }
  if (selectedVoice) utterance.voice = selectedVoice;

  utterance.rate   = voiceSettings.rate;   // 1.05
  utterance.pitch  = voiceSettings.pitch;  // 0.9
  utterance.volume = voiceSettings.volume; // 1.0

  utterance.onstart = () => {
    setOrbStatus('speaking');  // orb goes cyan, fast rings
  };

  utterance.onend = () => {
    setOrbStatus('online');    // orb returns to idle
    // Restart wake word listener after response finishes
    if (wakeWordActive) {
      setTimeout(() => startWakeWordListener(), 500);
    }
  };

  utterance.onerror = () => {
    setOrbStatus('online');
    if (wakeWordActive) startWakeWordListener();
  };

  // Chrome bug fix: speechSynthesis pauses after ~15s
  // Keep it alive with this interval
  const keepAlive = setInterval(() => {
    if (window.speechSynthesis.speaking) {
      window.speechSynthesis.pause();
      window.speechSynthesis.resume();
    } else {
      clearInterval(keepAlive);
    }
  }, 10000);

  window.speechSynthesis.speak(utterance);
}

VOICES LOAD TIMING FIX
──────────────────────────────────────────────────────

SpeechSynthesis voices load asynchronously in Chrome.
Always load voices like this, never synchronously:

  function getVoices(): Promise<SpeechSynthesisVoice[]> {
    return new Promise((resolve) => {
      const voices = window.speechSynthesis.getVoices();
      if (voices.length > 0) {
        resolve(voices);
      } else {
        window.speechSynthesis.onvoiceschanged = () => {
          resolve(window.speechSynthesis.getVoices());
        };
      }
    });
  }

Call this once on mount, store result in a ref.
Use the cached ref in speakResponse — never call getVoices() inline.
