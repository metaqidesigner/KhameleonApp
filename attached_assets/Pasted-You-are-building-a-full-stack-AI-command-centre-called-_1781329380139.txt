BACKEND  (/backend)
═══════════════════════════════════════════════════════

1. Clone and install OpenJarvis as a dependency:
     pip install git+https://github.com/open-jarvis/OpenJarvis.git

2. Create main.py — a FastAPI app that:
   a) Starts the OpenJarvis Jarvis() instance on app startup:
        from openjarvis import Jarvis
        jarvis = Jarvis()

   b) Exposes these REST endpoints:

      POST /api/ask
        Body: { "prompt": string, "agent": string?, "tools": string[]? }
        Response: { "content": string, "tool_results": any[], "model": string }

      GET  /api/agents
        → returns list of available OpenJarvis agent types:
          ["simple", "orchestrator", "deep_research", "morning_digest",
           "code_assistant", "channel_agent", "proactive_agent", "operative"]

      GET  /api/health
        → { "status": "online", "engine": string, "model": string, "uptime": number }

      GET  /api/memory/search?q=<query>
        → returns relevant memory chunks

      POST /api/memory/index
        Body: { "path": string }

      GET  /api/connectors
        → list of available connectors
          (gmail, gcalendar, slack, notion, gdrive, github, discord, telegram, etc.)

      POST /api/connectors/connect
        Body: { "connector": string }
        → initiates OAuth / connection flow

      POST /api/research
        Body: { "query": string }
        → uses agent="deep_research", returns streamed markdown

      GET  /api/mode
        → { "name": string } — current Jarvis operating mode

      GET  /api/skills
        → list of installed OpenJarvis skills

      POST /api/skills/install
        Body: { "source": string }
        → installs a skill e.g. "hermes:arxiv"

      WebSocket /ws/chat
        → streams tokens from Jarvis as generated
          client sends: { "prompt": string, "agent": string? }
          server emits: { "token": string } chunks, then { "done": true, "model": string }

   c) CORS: allow all origins in development

3. requirements.txt:
     fastapi
     uvicorn[standard]
     git+https://github.com/open-jarvis/OpenJarvis.git
     python-dotenv
     websockets

4. .env.example:
     OPENAI_API_KEY=
     ANTHROPIC_API_KEY=
     JARVIS_ENGINE=ollama
     JARVIS_MODEL=llama3.2

export function streamChat(prompt, agent, onToken, onDone) {
  const ws = new WebSocket(`${BASE.replace('http','ws')}/ws/chat`);
  ws.onopen = () => ws.send(JSON.stringify({ prompt, agent }));
  ws.onmessage = (e) => {
    const d = JSON.parse(e.data);
    if (d.token) onToken(d.token);
    if (d.done)  { onDone(d.model); ws.close(); }
  };
  ws.onerror = () => { onDone(null); };
  return () => ws.close();
}
